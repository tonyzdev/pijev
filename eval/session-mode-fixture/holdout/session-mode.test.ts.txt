type PlannedTool = { name: string; args: Record<string, unknown> };
function deferred<T>() { let resolve!: (value: T) => void; const promise = new Promise<T>((done) => { resolve = done; }); return { promise, resolve }; }
async function within<T>(promise: Promise<T>, message: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { return await Promise.race([promise, new Promise<never>((_, reject) => { timer = setTimeout(() => reject(new Error(message)), 1500); })]); }
  finally { clearTimeout(timer); }
}

// Only the model/Jev HTTP endpoints and UI sink are fakes. Commands, extension
// dispatch, sessions, state storage, branch navigation and replacements are SDK.
async function fixture(t: test.TestContext, initialMode: DecisionMode = "assist", withSkills = false) {
  const root = await mkdtemp(join(tmpdir(), "pijev-sm-fixture-"));
  const cwd = join(root, "project"), home = join(root, "agent"), sessionDir = join(root, "sessions");
  await mkdir(cwd);
  await writeFile(join(cwd, "a.ts"), "export const token = 'unrelated';\n");
  await writeFile(join(cwd, "b.ts"), "export function refreshToken(token: string) { return token; }\n");
  const skillPath = join(root, "SKILL.md");
  await writeFile(skillPath, "Inspect token refresh before editing. MODE_PRIVATE_SKILL\n");
  const modelRequests: string[] = [], notices: string[] = [], statuses: string[] = [], errors: string[] = [];
  const plan: PlannedTool[] = [];
  let defaultMode = initialMode, jevCalls = 0, modelCalls = 0, protocolError: unknown;
  let hold: { entered: ReturnType<typeof deferred<ServerResponse>>; closed: ReturnType<typeof deferred<void>> } | undefined;
  const http = createServer(async (req, res) => {
    try {
      let raw = ""; for await (const chunk of req) raw += String(chunk);
      if (req.url === "/v1/systemone") {
        jevCalls++;
        if (hold) {
          const pending = hold; hold = undefined;
          res.once("close", () => pending.closed.resolve());
          pending.entered.resolve(res);
          return;
        }
        const body = JSON.parse(raw) as { questions: Record<string, { type: string; criteria?: Record<string, unknown> }>; state: { candidates?: Record<string, { path?: string }> } };
        const answers = Object.fromEntries(Object.entries(body.questions).map(([id, question]) => {
          if (question.type === "noul") return [id, { type: "noul", noul: id.startsWith("c") ? body.state.candidates?.[id]?.path === "b.ts" ? 0.95 : 0.1 : 0.95 }];
          const keys = Object.keys(question.criteria ?? {});
          const choice = id === "category" && keys.includes("environment") ? "environment" : keys[0]!;
          return [id, { type: "choice", choice, probabilities: Object.fromEntries(keys.map((key) => [key, key === choice ? 1 : 0])), confidence: 0.95 }];
        }));
        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(JSON.stringify({ model: "mode-fixture", answers, usage: { input_tokens: 11, output_tokens: 0 } }));
        return;
      }
      assert.equal(req.url, "/v1/chat/completions");
      modelRequests.push(raw); modelCalls++;
      assert.ok(modelCalls < 80, "unexpected model loop");
      const next = plan.shift();
      const delta = next ? { role: "assistant", tool_calls: [{ index: 0, id: `mode_call_${modelCalls}`, type: "function", function: { name: next.name, arguments: JSON.stringify(next.args) } }] } : { role: "assistant", content: "Local fixture complete." };
      res.writeHead(200, { "Content-Type": "text/event-stream" });
      for (const [part, finish] of [[delta, null], [{}, next ? "tool_calls" : "stop"]]) res.write(`data: ${JSON.stringify({ id: `local_${modelCalls}`, object: "chat.completion.chunk", model: "fixture", choices: [{ index: 0, delta: part, finish_reason: finish }] })}\n\n`);
      res.end("data: [DONE]\n\n");
    } catch (error) { protocolError = error; res.writeHead(500); res.end("local fixture protocol error"); }
  });
  http.listen(0, "127.0.0.1"); await once(http, "listening");
  const address = http.address(); assert.ok(address && typeof address === "object");
  const baseUrl = `http://127.0.0.1:${address.port}/v1`;
  const modelRuntime = await ModelRuntime.create({ authPath: join(home, "auth.json"), modelsPath: null, refreshOnCreate: false });
  modelRuntime.registerProvider("mode-local", { baseUrl, api: "openai-completions", apiKey: "MODE_PRIVATE_MODEL_KEY", models: [{ id: "fixture", name: "fixture", reasoning: false, input: ["text"], cost: { input: 0, output: 0, cacheRead: 0, cacheWrite: 0 }, contextWindow: 32000, maxTokens: 512 }] });
  const settings = SettingsManager.inMemory({ retry: { enabled: false }, compaction: { enabled: false } });
  const bind = (session: AgentSession) => session.bindExtensions({
    mode: "json", onError: (error) => errors.push(`${error.event}: ${error.error}`),
    uiContext: { ...session.extensionRunner.createContext().ui, notify: (message) => notices.push(message), setStatus: (_key, message) => { if (message) statuses.push(message); } },
  });
  let lastResources: ResourceLoader | undefined;
  let reuseResources = false;
  const factory: CreateAgentSessionRuntimeFactory = async (options) => {
    let resources = reuseResources ? lastResources : undefined;
    if (!resources) {
      resources = new DefaultResourceLoader({
        cwd, agentDir: home, settingsManager: settings, noExtensions: true, noSkills: true, noPromptTemplates: true, noThemes: true,
        agentsFilesOverride: () => ({ agentsFiles: [] }),
        skillsOverride: () => ({ skills: withSkills ? [{ name: "tokens", description: "Inspect tokens", filePath: skillPath, baseDir: root, sourceInfo: { path: skillPath, source: "fixture", scope: "temporary", origin: "top-level" }, disableModelInvocation: false }] : [], diagnostics: [] }),
        extensionFactories: [createPijevExtension(loadConfig({ PIJEV_HOME: home, PIJEV_MODE: defaultMode, TYPESAFE_API_KEY: "MODE_PRIVATE_JEV_KEY", PIJEV_JEV_ENDPOINT: `${baseUrl}/systemone`, PIJEV_JEV_TIMEOUT_MS: "5000" }))],
      });
      await resources.reload(); assert.deepEqual(resources.getExtensions().errors, []);
      lastResources = resources;
    }
    const result = await createAgentSession({ ...options, modelRuntime, model: modelRuntime.getModel("mode-local", "fixture"), settingsManager: settings, resourceLoader: resources, tools: ["pijev_search", "bash"] });
    return { ...result, services: { cwd, agentDir: home, settingsManager: settings, modelRuntime, resourceLoader: resources, diagnostics: [] }, diagnostics: [] };
  };
  const runtime = await createAgentSessionRuntime(factory, { cwd, agentDir: home, sessionManager: SessionManager.create(cwd, sessionDir) });
  runtime.setRebindSession(bind); await bind(runtime.session);
  t.after(async () => {
    http.closeAllConnections(); await runtime.session.abort(); await runtime.dispose();
    await new Promise<void>((resolve) => http.close(() => resolve()));
    await rm(root, { recursive: true, force: true });
    assert.ifError(protocolError); assert.deepEqual(errors, [], "extension runtime errors");
  });
  return {
    runtime, notices, statuses, modelRequests, home,
    get jevCalls() { return jevCalls; }, get modelCalls() { return modelCalls; },
    setDefault(mode: DecisionMode) { defaultMode = mode; },
    reuseExtensions() { reuseResources = true; },
    async select(mode: DecisionMode) { await runtime.session.prompt(`/pijev ${mode}`); },
    async assertMode(mode: DecisionMode) { await runtime.session.prompt("/pijev status"); assert.match(notices.at(-1)!, new RegExp(`Mode: ${mode}\\b`)); },
    async save(text = "Save MODE_PRIVATE_PROMPT to the local session.") { await runtime.session.prompt(text); const leaf = runtime.session.sessionManager.getLeafId(); assert.ok(leaf); return leaf; },
    async reopen() { const path = runtime.session.sessionFile; assert.ok(path); assert.equal((await runtime.switchSession(path)).cancelled, false); },
    searchAndFail() { plan.push({ name: "pijev_search", args: { query: "Find refresh token implementation", patterns: ["token"] } }, { name: "bash", args: { command: "exit 2" } }); },
    blockNextDecision() { hold = { entered: deferred<ServerResponse>(), closed: deferred<void>() }; return hold; },
  };
}

const records = (session: AgentSession) => session.sessionManager.getEntries().filter((entry) => entry.type === "custom" && entry.customType === "pijev-mode");

test("session mode: disk reopen restores all explicit selections over a different configured default", { timeout: 15000 }, async (t) => {
  const h = await fixture(t, "off");
  for (const chosen of ["observe", "assist", "off"] as const) {
    await h.select(chosen); await h.save();
    const file = h.runtime.session.sessionFile; assert.ok(file);
    const data = (await readFile(file, "utf8")).trim().split("\n").map((line) => JSON.parse(line));
    assert.deepEqual(data.filter((entry) => entry.type === "custom" && entry.customType === "pijev-mode").at(-1)?.data, { version: 1, mode: chosen });
    h.setDefault(chosen === "off" ? "assist" : "off"); await h.reopen(); await h.assertMode(chosen);
  }
});

test("session mode: tree navigation restores ancestors and divergent branches immediately", { timeout: 15000 }, async (t) => {
  const h = await fixture(t, "observe");
  const root = await h.save();
  await h.select("off"); const off = await h.save();
  await h.select("assist"); const assist = await h.save();
  assert.equal((await h.runtime.session.navigateTree(off, { summarize: false })).cancelled, false);
  // No /pijev command may lazily repair mode state before this next actual turn.
  const calls = h.jevCalls;
  h.searchAndFail(); await h.save("Search immediately after tree navigation to off.");
  assert.equal(h.jevCalls, calls, "tree navigation must make off effective before the next command/status read");
  const search = h.runtime.session.messages.findLast((message) => message.role === "toolResult" && message.toolName === "pijev_search");
  assert.ok(search && search.role === "toolResult" && !search.isError);
  assert.ok(JSON.stringify(search).includes("lexical order"));
  const failure = h.runtime.session.messages.findLast((message) => message.role === "toolResult" && message.toolName === "bash");
  assert.ok(failure && failure.role === "toolResult" && failure.isError);
  assert.ok(!JSON.stringify(failure).includes("PiJev diagnostic suggestion"));
  await h.assertMode("off");
  await h.select("observe"); const sibling = await h.save();
  const total = records(h.runtime.session).length;
  for (const [leaf, mode] of [[assist, "assist"], [root, "observe"], [sibling, "observe"], [off, "off"]] as const) {
    await h.runtime.session.navigateTree(leaf, { summarize: false }); await h.assertMode(mode);
  }
  assert.equal(records(h.runtime.session).length, total, "reading/navigating must not append mode selections");
});

test("session mode: public runtime fork inherits selected ancestry and keeps parent/fork/new sessions independent", { timeout: 15000 }, async (t) => {
  const h = await fixture(t, "assist");
  await h.save(); await h.select("observe"); const forkPoint = await h.save();
  await h.select("off"); await h.save(); const parent = h.runtime.session.sessionFile; assert.ok(parent);
  assert.equal((await h.runtime.fork(forkPoint, { position: "at" })).cancelled, false); await h.assertMode("observe");
  await h.select("assist"); await h.save(); const child = h.runtime.session.sessionFile; assert.ok(child); assert.notEqual(child, parent);
  await h.runtime.switchSession(parent); await h.assertMode("off");
  await h.runtime.newSession(); await h.assertMode("assist");
  await h.runtime.switchSession(child); await h.assertMode("assist");
  await h.runtime.switchSession(parent); await h.assertMode("off");
});

test("session mode: malformed, extra-field, unknown and future entries are ignored without hiding older valid state", { timeout: 15000 }, async (t) => {
  const h = await fixture(t, "assist"); await h.save();
  const invalid: unknown[] = [null, [], "off", {}, { mode: "off" }, { version: 1 }, { version: "1", mode: "off" }, { version: 0, mode: "off" }, { version: 2, mode: "off" }, { version: 1, mode: "OFF" }, { version: 1, mode: "other" }, { version: 1, mode: null }, { version: 1, mode: "off", prompt: "MODE_INVALID_PRIVATE" }];
  const manager = h.runtime.session.sessionManager;
  for (const value of invalid) manager.appendCustomEntry("pijev-mode", value);
  manager.appendCustomEntry("unrelated-mode", { version: 1, mode: "off" });
  await h.reopen(); await h.assertMode("assist");
  await h.select("observe");
  for (const value of invalid) h.runtime.session.sessionManager.appendCustomEntry("pijev-mode", value);
  h.runtime.session.sessionManager.appendCustomEntry("unrelated-mode", { version: 1, mode: "off" });
  await h.reopen(); await h.assertMode("observe");
});

test("session mode: allowlisted custom metadata never enters the model context or captures private input", { timeout: 15000 }, async (t) => {
  const h = await fixture(t);
  await h.save("MODE_PRIVATE_PROMPT source text and arbitrary arguments");
  for (const mode of ["assist", "observe", "off"] as const) await h.select(mode);
  const before = records(h.runtime.session);
  assert.equal(before.length, 3);
  assert.deepEqual(before.map((entry) => entry.type === "custom" ? entry.data : undefined), [{ version: 1, mode: "assist" }, { version: 1, mode: "observe" }, { version: 1, mode: "off" }]);
  for (const entry of before) {
    assert.equal(entry.type, "custom");
    if (entry.type !== "custom") throw new Error("expected SDK custom entry");
    assert.ok(Buffer.byteLength(JSON.stringify(entry.data)) <= 64);
    assert.ok(!JSON.stringify(entry.data).includes("MODE_PRIVATE"));
  }
  for (const command of ["/pijev status", "/pijev decisions", "/pijev off MODE_PRIVATE_ARGUMENT"]) await h.runtime.session.prompt(command);
  assert.equal(records(h.runtime.session).length, before.length, "status, decisions and invalid commands must not persist selections");
  await h.reopen(); await h.save("Check model context after resume.");
  const transcript = h.modelRequests.join("\n");
  assert.ok(!transcript.includes("pijev-mode"), "custom mode state is not a model message");
  for (const sentinel of ["MODE_PRIVATE_ARGUMENT", "MODE_PRIVATE_MODEL_KEY", "MODE_PRIVATE_JEV_KEY"]) assert.ok(!transcript.includes(sentinel));
  const messages = h.runtime.session.sessionManager.buildSessionContext().messages;
  assert.ok(!JSON.stringify(messages).includes("pijev-mode"));
});

test("session mode: restored modes preserve ranking, skill advice and triage effects; restored off makes zero Jev calls", { timeout: 20000 }, async (t) => {
  const h = await fixture(t, "off", true);
  for (const mode of ["assist", "observe", "off"] as const) {
    await h.select(mode); await h.save();
    h.setDefault(mode === "off" ? "assist" : "off"); await h.reopen();
    const calls = h.jevCalls, modelStart = h.modelRequests.length;
    h.searchAndFail(); await h.save(`Inspect token refresh for restored ${mode}.`);
    const messages = h.runtime.session.messages;
    const search = messages.findLast((message) => message.role === "toolResult" && message.toolName === "pijev_search");
    assert.ok(search && search.role === "toolResult" && !search.isError);
    const text = JSON.stringify(search);
    assert.ok(text.includes("a.ts") && text.includes("b.ts"));
    assert.equal(text.includes("Jev ranked"), mode === "assist");
    assert.equal(text.indexOf("b.ts") < text.indexOf("a.ts"), mode === "assist");
    const failure = messages.findLast((message) => message.role === "toolResult" && message.toolName === "bash");
    assert.ok(failure && failure.role === "toolResult" && failure.isError);
    assert.equal(JSON.stringify(failure).includes("PiJev diagnostic suggestion"), mode === "assist");
    const lastRequest = JSON.parse(h.modelRequests[modelStart]!) as { messages: { role: string; content: unknown }[] };
    // The last user prompt splits retained transcript history from this turn's advice.
    const lastUser = lastRequest.messages.findLastIndex((message) => message.role === "user" && JSON.stringify(message.content).includes(`restored ${mode}`));
    assert.ok(lastUser >= 0);
    assert.equal(JSON.stringify(lastRequest.messages.slice(lastUser)).includes("PiJev skill suggestion"), mode === "assist");
    assert.equal(h.jevCalls - calls, mode === "off" ? 0 : 4);
  }
});

test("session mode: a real mode command aborts an in-flight decision and keeps off effective after reopen", { timeout: 15000 }, async (t) => {
  const h = await fixture(t, "assist", true);
  const blocked = h.blockNextDecision();
  const running = h.save("Inspect token refresh while a decision is pending.");
  await within(blocked.entered.promise, "skill request did not arrive");
  await h.select("off");
  await within(blocked.closed.promise, "mode command must abort the held Jev HTTP request before its deadline");
  await within(running, "mode change did not let the current turn finish promptly");
  assert.equal(h.jevCalls, 1);
  assert.ok(!h.modelRequests.join("\n").includes("PiJev skill suggestion"), "aborted advice must not appear later");
  await h.assertMode("off"); await h.reopen(); await h.assertMode("off");
  const calls = h.jevCalls;
  h.searchAndFail(); await h.save("Continue after restoring off.");
  assert.equal(h.jevCalls, calls, "off after restore must not launch a new decision");

  // Exercise the separate search-ranking signal path with no eligible skills.
  const searchCase = await fixture(t, "assist", false);
  const ranking = searchCase.blockNextDecision();
  searchCase.searchAndFail();
  const searching = searchCase.save("Search while a ranking decision is pending.");
  await within(ranking.entered.promise, "ranking request did not arrive");
  await searchCase.select("off");
  await within(ranking.closed.promise, "mode command must abort the held search-ranking HTTP request");
  await within(searching, "cancelled ranking must let the tool turn finish promptly");
  assert.equal(searchCase.jevCalls, 1, "off must suppress subsequent failure triage");
  const result = searchCase.runtime.session.messages.findLast((message) => message.role === "toolResult" && message.toolName === "pijev_search");
  assert.ok(result && result.role === "toolResult" && !result.isError);
  assert.ok(JSON.stringify(result).includes("lexical order"));
  assert.ok(!JSON.stringify(result).includes("Jev ranked"), "cancelled ranking must not be applied");
  await searchCase.reopen(); await searchCase.assertMode("off");
});
